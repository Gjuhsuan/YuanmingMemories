常用查询
───────
  在 Neo4j Browser (http://localhost:7474) 中粘贴执行。

  搜索实体
    MATCH (n) WHERE n.name CONTAINS "雍正" RETURN n.name, labels(n), n.entity_type_cn LIMIT 20

  搜索事件
    MATCH (e:Event) WHERE e.name CONTAINS "圆明园" RETURN e.name, e.event_type_cn, e.date_text LIMIT 20

  某人参与的事件
    MATCH (p) WHERE p.name CONTAINS "雍正"
    MATCH (p)<-[r]-(e:Event)
    RETURN e.name, type(r), e.date_text, r.relation_cn ORDER BY e.date_text LIMIT 30

  某地发生的事件
    MATCH (pl:Place) WHERE pl.name CONTAINS "圆明园"
    MATCH (pl)<-[:INVOLVES_PLACE]-(e:Event)
    RETURN e.name, e.event_type_cn, e.date_text LIMIT 30

  事件详情（含全部关联）
    MATCH (e:Event {event_id: "evt_21_1"})
    OPTIONAL MATCH (e)-[r]-(n)
    RETURN e, r, n

  两实体最短路径
    MATCH p = shortestPath((a)-[*1..3]-(b))
    WHERE a.name CONTAINS "雍正" AND b.name CONTAINS "圆明园"
    RETURN [n in nodes(p) | n.name] AS 路径,
           [r in relationships(p) | type(r)] AS 关系

  按事件类型查询
    MATCH (e:Event {event_type_cn: "营造事件"})
    RETURN e.name, e.date_text LIMIT 20

  人物关系网络
    MATCH (p) WHERE p.name CONTAINS "雍正"
    MATCH (p)-[r]-(m)
    RETURN type(r), r.relation_cn, m.name, m.entity_type_cn LIMIT 50

  某人官职
    MATCH (p) WHERE p.name CONTAINS "马齐"
    MATCH (p)-[:HOLDS_TITLE]->(t)
    RETURN p.name, t.name

  事件因果关系
    MATCH (e1:Event)-[:CAUSES]->(e2:Event)
    RETURN e1.name, e2.name LIMIT 30

  from neo4j import GraphDatabase

  d = GraphDatabase.driver("bolt://localhost:7687", auth=("neo4j", "ymysj123"))
  with d.session(database="neo4j") as s:
      for r in s.run("MATCH (e:Event) WHERE e.name CONTAINS '圆明园' RETURN e LIMIT 5").data():
          print(r)
  d.close()
